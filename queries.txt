-- Table structure for table Room_Category
CREATE TABLE Room_Category (
  room_category_id SERIAL PRIMARY KEY,
  category VARCHAR(45) NULL
);

-- Table structure for table Hotel
CREATE TABLE Hotel (
  hotel_id SERIAL PRIMARY KEY,
  name VARCHAR(45) NULL,
  email VARCHAR(45) NULL,
  phone VARCHAR(10) NULL,
  country VARCHAR(45) NULL,
  street VARCHAR(45) NULL,
  city VARCHAR(45) NULL
);

-- Table structure for table Room
CREATE TABLE Room (
  room_id SERIAL PRIMARY KEY,
  price FLOAT NULL,
  description TEXT NULL,
  condition VARCHAR(45) NULL,
  hotel_id INT NOT NULL,
  room_category_id INT NOT NULL,
  CONSTRAINT fk_Room_Hotel1 FOREIGN KEY (hotel_id) REFERENCES Hotel (hotel_id),
  CONSTRAINT fk_Room_Room_Category1 FOREIGN KEY (room_category_id) REFERENCES Room_Category (room_category_id)
);

-- Table structure for table Facility
CREATE TABLE Facility (
  facility_id SERIAL PRIMARY KEY,
  facility VARCHAR(45) NULL,
  hotel_id INT NOT NULL,
  CONSTRAINT fk_Facility_Hotel1 FOREIGN KEY (hotel_id) REFERENCES Hotel (hotel_id)
);

-- Table structure for table Transaction
CREATE TABLE Transaction (
  transaction_id SERIAL PRIMARY KEY,
  type VARCHAR(45) NULL,
  date DATE NULL,
  time TIME NULL,
  amount FLOAT NULL
);

-- Table structure for table Traveler
CREATE TABLE Traveler (
  traveler_id SERIAL PRIMARY KEY,
  first_name VARCHAR(45) NULL,
  last_name VARCHAR(45) NULL,
  country VARCHAR(45) NULL,
  address VARCHAR(45) NULL,
  email VARCHAR(45) NULL,
  phone VARCHAR(12) NULL
);

-- Table structure for table Reservation
CREATE TABLE Reservation (
  reservation_id SERIAL PRIMARY KEY,
  date DATE NULL,
  duration VARCHAR(45) NULL,
  no_of_guests VARCHAR(45) NULL,
  time TIME NULL,
  amount_paid FLOAT NULL,
  traveler_id INT NOT NULL,
  transaction_id INT NOT NULL,
  room_id INT NOT NULL,
  CONSTRAINT fk_Reservation_Traveler FOREIGN KEY (traveler_id) REFERENCES Traveler (traveler_id),
  CONSTRAINT fk_Reservation_Transaction1 FOREIGN KEY (transaction_id) REFERENCES Transaction (transaction_id),
  CONSTRAINT fk_Reservation_Room1 FOREIGN KEY (room_id) REFERENCES Room (room_id)
);

CREATE TABLE Report (
  report_id SERIAL PRIMARY KEY,
  title VARCHAR(45) NULL,
  description VARCHAR(45) NULL,
  date DATE NULL,
  country VARCHAR(45) NULL,
  location VARCHAR(45) NULL,
  traveler_id INT NOT NULL,
  CONSTRAINT fk_Report_Traveler1 FOREIGN KEY (traveler_id) REFERENCES Traveler (traveler_id)
);
